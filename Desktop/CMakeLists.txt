cmake_minimum_required(VERSION 3.20)
project(SieuvietDesktop LANGUAGES C CXX)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../Design/brand.json" SIEUVIET_BRAND)
string(JSON SIEUVIET_VERSION GET "${SIEUVIET_BRAND}" version)
string(JSON SIEUVIET_NAME GET "${SIEUVIET_BRAND}" name)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../Design/content.json" SIEUVIET_CONTENT)
string(JSON SIEUVIET_SETTINGS_VI GET "${SIEUVIET_CONTENT}" keyboardSettings vi)
string(JSON SIEUVIET_SETTINGS_EN GET "${SIEUVIET_CONTENT}" keyboardSettings en)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(GNUInstallDirs)
include(CTest)
add_library(sieuviet-desktop INTERFACE)
target_include_directories(sieuviet-desktop INTERFACE common ../Sources/CABI/include)
if(WIN32)
  if(MINGW)
    target_link_options(sieuviet-desktop INTERFACE -static)
  endif()
  if(MSVC)
    target_compile_options(sieuviet-desktop INTERFACE /utf-8 /W4 /permissive-)
  endif()
  target_compile_definitions(sieuviet-desktop INTERFACE UNICODE _UNICODE NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
  add_subdirectory(windows)
else()
  target_link_libraries(sieuviet-desktop INTERFACE ${CMAKE_DL_LIBS})
  option(SIEUVIET_LINUX_ADAPTERS "Build IBus and Fcitx 5 adapters" ON)
  if(SIEUVIET_LINUX_ADAPTERS AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_subdirectory(linux)
  endif()
endif()
if(BUILD_TESTING)
  add_executable(desktop-session-tests tests/session.cpp)
  target_link_libraries(desktop-session-tests PRIVATE sieuviet-desktop)
  add_test(NAME desktop-session COMMAND desktop-session-tests)
endif()
