find_package(PkgConfig REQUIRED)
pkg_check_modules(IBUS REQUIRED IMPORTED_TARGET ibus-1.0)
pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0)
find_package(Fcitx5Core REQUIRED)
find_package(Qt5Widgets REQUIRED)
set(SIEUVIET_ENGINE_PATH "${CMAKE_INSTALL_FULL_LIBDIR}/sieuviet/libSieuvietEngine.so" CACHE FILEPATH "Installed shared Swift engine")
add_executable(ibus-engine-sieuviet ibus.cpp)
target_link_libraries(ibus-engine-sieuviet PRIVATE sieuviet-desktop PkgConfig::IBUS)
target_compile_definitions(ibus-engine-sieuviet PRIVATE SIEUVIET_ENGINE_PATH="${SIEUVIET_ENGINE_PATH}")
add_library(sieuviet MODULE fcitx.cpp)
target_compile_features(sieuviet PRIVATE cxx_std_20)
target_link_libraries(sieuviet PRIVATE sieuviet-desktop Fcitx5::Core)
target_compile_definitions(sieuviet PRIVATE SIEUVIET_ENGINE_PATH="${SIEUVIET_ENGINE_PATH}")
install(TARGETS ibus-engine-sieuviet RUNTIME DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/sieuviet)
install(TARGETS sieuviet LIBRARY DESTINATION ${FCITX_INSTALL_ADDONDIR})
configure_file(ibus.xml.in sieuviet.xml @ONLY)
configure_file(org.sieuviet.Settings.desktop.in org.sieuviet.Settings.desktop @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/sieuviet.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/ibus/component)
configure_file(fcitx-addon.conf ${CMAKE_CURRENT_BINARY_DIR}/fcitx-addon.conf @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/fcitx-addon.conf DESTINATION ${CMAKE_INSTALL_DATADIR}/fcitx5/addon RENAME sieuviet.conf)
configure_file(fcitx-input.conf ${CMAKE_CURRENT_BINARY_DIR}/fcitx-input.conf @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/fcitx-input.conf DESTINATION ${CMAKE_INSTALL_DATADIR}/fcitx5/inputmethod RENAME sieuviet.conf)
install(PROGRAMS settings.py DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME sieuviet-settings)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/org.sieuviet.Settings.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES ../common/design.json DESTINATION ${CMAKE_INSTALL_DATADIR}/sieuviet)
add_executable(linux-host-test host-test.cpp)
target_link_libraries(linux-host-test PRIVATE PkgConfig::GTK3)

add_executable(linux-qt-host-test qt-host-test.cpp)
target_link_libraries(linux-qt-host-test PRIVATE Qt5::Widgets)

install(FILES ../common/icon.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps RENAME sieuviet.svg)
